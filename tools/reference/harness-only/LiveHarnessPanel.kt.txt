package com.projectrevivesolutions.layertime.controller.ui
import androidx.compose.material3.Text
import androidx.compose.runtime.*
import androidx.compose.ui.unit.sp
import com.projectrevivesolutions.layertime.controller.signatures.LiveHarness
import com.projectrevivesolutions.layertime.controller.ui.theme.LtColors
import kotlinx.coroutines.delay
@Composable internal fun LiveHarnessPanel(onDismiss:()->Unit) {
    val state by LiveHarness.state.collectAsState()
    var elapsed by remember {mutableStateOf(0L)}
    LaunchedEffect(state.aiStartedMs){while(state.aiStartedMs!=null){elapsed=(android.os.SystemClock.elapsedRealtime()-state.aiStartedMs!!)/1000;delay(1000)}}
    BuilderModal(onDismiss) {
        Text("LIVE HUNTER TEST",color=LtColors.Cyan,fontSize=19.sp)
        Text("Button-driven source - quarantined observations",color=LtColors.Grey,fontSize=12.sp)
        Text("Connect Hunter-Test, then press its BOOT button. The dongle sends the sequence in real time.",color=LtColors.Ivory,fontSize=14.sp)
        Text(state.phase,color=LtColors.Amber,fontSize=18.sp)
        Text("Source t+${state.sourceSeconds}s - ${state.preserved} slices preserved",color=LtColors.Cyan,fontSize=14.sp)
        Text("Dongle SD: ${state.sourceSd}",color=LtColors.Cyan,fontSize=13.sp)
        state.timeline.forEach {Text(it,color=LtColors.Ivory,fontSize=13.sp)}
        state.classification?.let {Text(it,color=LtColors.Amber,fontSize=22.sp)}
        if(state.aiStartedMs!=null)Text("Offline AI reading the preserved evidence... ${elapsed}s",color=LtColors.Cyan,fontSize=14.sp)
        state.answer?.let {a->Text("AI: ${a.headline}",color=LtColors.Amber,fontSize=18.sp)
            Text(a.significance,color=LtColors.Ivory,fontSize=15.sp);Text(a.limitations,color=LtColors.Grey,fontSize=12.sp)}
        state.problem?.let {Text(it,color=LtColors.Amber,fontSize=14.sp)}
        Text("Source telemetry is generated by the test dongle. The phone uses its real preservation, matching and offline AI pipeline. No test label is sent to the model.",color=LtColors.Grey,fontSize=12.sp)
    }
}
