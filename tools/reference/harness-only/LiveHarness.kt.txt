package com.projectrevivesolutions.layertime.controller.signatures
import android.content.Context
import android.os.SystemClock
import com.projectrevivesolutions.layertime.controller.BuildConfig
import com.projectrevivesolutions.layertime.controller.ai.*
import com.projectrevivesolutions.layertime.controller.ai.runtime.BundledModelFactory
import com.projectrevivesolutions.layertime.controller.flow.FlowJson
import kotlinx.coroutines.*
import kotlinx.coroutines.channels.Channel
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asStateFlow
import java.io.File
import java.security.MessageDigest

/** Live BLE controlled source; payloads remain separate from field collections. */
object LiveHarness {
    data class State(val peer:String?=null,val session:Int=0,val run:String="",val phase:String="Connect Hunter-Test, then press BOOT",
        val sourceSeconds:Long=0,val timeline:List<String> = emptyList(),val preserved:Int=0,val classification:String?=null,
        val aiStartedMs:Long?=null,val answer:FieldShadowTasks.Explanation?=null,val problem:String?=null,val sourceSd:String="Unknown")
    private val mutable=MutableStateFlow(State());val state=mutable.asStateFlow()
    private val scope=CoroutineScope(SupervisorJob()+Dispatchers.IO)
    private val input=Channel<Triple<Context,Pair<String,Int>,ByteArray?>>(256)
    private class Run(val context:Context,val peer:String,val session:Int,val id:Long) {
        val key=java.util.UUID.randomUUID().toString()
        val clock="source-clock:$peer/$session/$id"
        val dir=File(context.filesDir,"quarantined-harness/$key").apply {check(mkdirs())}
        var last=0;var lastTime=0L;var stage=0;var baseline=false;var bad=false;var startedAi=false
        val closed=linkedMapOf<String,Map<String,Any?>>()
        val preserver=LiveEvidenceSlices({snapshot->
            File(dir,"slices.jsonl").appendText(FlowJson.write(mapOf("quarantined" to true,"fields" to snapshot))+"\n")
            if(snapshot["state"]=="CLOSED")closed[snapshot["sliceId"] as String]=snapshot
        })
    }
    private var run:Run?=null
    init {scope.launch {for((context,source,bytes) in input) {
        try {
            if(bytes==null) {
                val current=run
                if(current!=null && current.peer==source.first && !current.startedAi) {
                    current.bad=true;mutable.value=mutable.value.copy(phase="Source disconnected",problem="Press BOOT for a new run after reconnecting.")
                }
            } else process(context,source.first,source.second,bytes)
        } catch(ex:Exception) {mutable.value=mutable.value.copy(problem="Run stopped: ${ex.message}",phase="Evidence incomplete");run?.bad=true}
    }}}
    fun connected(context:Context,peer:String,session:Int) {
        if(BuildConfig.DEBUG && (run==null || state.value.peer!=peer))
            mutable.value=State(peer=peer,session=session,phase="Connected - press the dongle BOOT button")
    }
    fun disconnected(peer:String?) {if(BuildConfig.DEBUG && peer!=null && state.value.peer==peer) {
        val c=run?.context ?: return
        input.trySend(Triple(c,peer to state.value.session,null))
    }}
    fun receive(context:Context,peer:String,session:Int,bytes:ByteArray) {
        if(!BuildConfig.DEBUG)return
        if(input.trySend(Triple(context.applicationContext,peer to session,bytes.copyOf())).isFailure)
            mutable.value=mutable.value.copy(phase="Receiver queue full",problem="Run incomplete; start a new run.")
    }
    private fun process(context:Context,peer:String,session:Int,bytes:ByteArray) {
        val s=HarnessProtocol.decode(bytes) ?: error("Invalid source packet")
        val old=run
        val r=if(old==null || old.id!=s.run || old.peer!=peer || old.session!=session) {
            check(s.sequence==1 && s.stage==1 && s.elapsedMs<2000) {"Start packet missing; press BOOT again."}
            Run(context,peer,session,s.run).also {run=it;mutable.value=State(peer,session,it.key,"Baseline activity",timeline=listOf("t+0s  Source button started the run"))}
        } else old
        File(r.dir,"received.jsonl").appendText(FlowJson.write(mapOf("quarantined" to true,"peer" to peer,"session" to session,
            "run" to s.run,"sequence" to s.sequence,"sourceElapsedMs" to s.elapsedMs,"receivedMonotonicMs" to SystemClock.elapsedRealtime(),
            "frameHex" to bytes.joinToString(""){"%02X".format(it)},"flags" to s.flags,"stage" to s.stage,"wifi" to s.wifi,"ble" to s.ble,"drone" to s.drone))+"\n")
        if(r.bad)return
        check(s.sequence==r.last+1 && s.elapsedMs>=r.lastTime && (r.last==0 || s.elapsedMs-r.lastTime<=2500)) {"BLE sequence or timing gap; start a new run."}
        r.last=s.sequence;r.lastTime=s.elapsedMs
        if(s.stage==1)r.baseline=s.wifi>0 && s.ble>0
        val transition=s.stage!=r.stage;r.stage=s.stage
        val kind=if(transition && s.stage==3)"wifi-ble-outage" else if(transition && s.stage==4)"drone-observation" else null
        val fields=linkedMapOf<String,Any?>("hunterAddress" to peer,"sessionId" to session,"sourceRun" to s.run)
        if(kind!=null)fields.putAll(mapOf("eventId" to if(s.stage==3)1L else 2L,"count" to 1,"confidence" to 2,"ageSeconds" to 0,
            "observationKind" to kind,"observedAtMs" to if(s.stage==3)20000L else 50000L,"producer" to "Hunter"))
        r.preserver.accept(LiveEvidenceSlices.Record(r.key,s.sequence.toLong(),s.elapsedMs,if(kind==null)"status" else "event_summary",
            fields,true,r.clock,"source_monotonic_elapsed_ms"))
        if(state.value.run==r.key) {
            val label=when(s.stage){1->"Baseline: Wi-Fi 120 / BLE 80";2->"Wi-Fi/BLE activity dropped";3->"Outage detected - preserving context";4->"Drone signature received";else->"Source sequence complete"}
            val timeline=if(transition)state.value.timeline+"t+${s.elapsedMs/1000}s  $label" else state.value.timeline
            mutable.value=state.value.copy(sourceSeconds=s.elapsedMs/1000,timeline=timeline,preserved=r.closed.size,
                sourceSd=if(s.flags and 8!=0) "Write failure" else if(s.flags and 4!=0) "Source record available" else "Not recording",
                phase=if(r.startedAi)"AI analyzing preserved evidence" else label)
        }
        if(r.closed.size==2 && !r.startedAi) {
            check(r.baseline) {"Baseline evidence missing"};r.startedAi=true
            val slices=r.closed.values.map {snapshot->
                val observations=(snapshot["records"] as List<*>).mapNotNull {v->
                    val row=v as Map<*,*>;val f=row["fields"] as Map<*,*>;val k=f["observationKind"] as? String
                    if(k==null)null else AttackSignatures.Observation("${r.key}:${row["sequence"]}",k,f["observedAtMs"] as Long,
                        f["producer"] as String,"${snapshot["sliceId"]}#${row["sequence"]}")
                }
                AttackSignatures.Slice(snapshot["sliceId"] as String,r.key,r.clock,snapshot["requestedStartMs"] as Long,
                    snapshot["requestedEndMs"] as Long,observations,listOf("Only the supplied source intervals are covered; continuous RF coverage is not established."))
            }
            val assessment=AttackSignatures.assess(AttackSignatures.Signature("evacuation-event",1,"wifi-ble-outage","drone-observation",90000),slices).single()
            val outage=EvacuationPolicy.Outage(120,0,80,0,true,true,0)
            val decision=EvacuationPolicy.evaluate(assessment,outage);check(decision.matched)
            mutable.value=state.value.copy(phase="AI analyzing preserved evidence",classification=decision.headline,aiStartedMs=SystemClock.elapsedRealtime(),
                timeline=state.value.timeline+"Rule matched: outage then drone in 30s; evidence preserved")
            analyze(r,assessment,outage,decision)
        }
    }
    private fun analyze(r:Run,a:AttackSignatures.Assessment,outage:EvacuationPolicy.Outage,d:EvacuationPolicy.Decision)=scope.launch {
        val provider=BundledModelFactory.development(r.context);val start=SystemClock.elapsedRealtime()
        try {
            val manifest=requireNotNull(provider.availability().manifest);check(!manifest.blueTeamTrained)
            val request=EvacuationPolicy.request(a,outage)
            check(request.messages.none {it.content.contains("quarantined",true) || it.content.contains("simulation",true)})
            check(AiCoordinator(provider).execute(request){it} is AiOutcome.Unavailable)
            val json=FlowJson.write(mapOf("quarantined" to true,"weightsSha256" to manifest.sha256,"modelId" to manifest.id,
                "messages" to request.messages.map {mapOf("role" to it.role.name,"content" to it.content)},"maxOutputTokens" to request.maxOutputTokens,"outputSchema" to request.outputSchema))
            File(r.dir,"model-input.json").writeText(json)
            val tokens=withTimeout(180000){provider.countInputTokens(request)};check(tokens+request.maxOutputTokens<=manifest.contextTokens)
            val response=withTimeout(300000){provider.generate(request)}
            File(r.dir,"model-output.json").writeText(FlowJson.write(mapOf("output" to response.json,"totalMs" to SystemClock.elapsedRealtime()-start,
                "inputSha256" to MessageDigest.getInstance("SHA-256").digest(json.toByteArray()).joinToString(""){"%02x".format(it)})))
            check(response.modelId==manifest.id && response.modelVersion==manifest.version && response.toolCalls.isEmpty())
            val answer=FieldShadowTasks.decode(response.json);check(EvacuationPolicy.reportProblems(d,answer).isEmpty()) {"AI changed the verified finding"}
            if(state.value.run==r.key)mutable.value=state.value.copy(phase="AI assessment complete",answer=answer,aiStartedMs=null,
                timeline=state.value.timeline+"AI explanation verified in ${(SystemClock.elapsedRealtime()-start)/1000}s")
        } catch(ex:Exception) {
            File(r.dir,"model-error.txt").writeText(ex.toString())
            if(state.value.run==r.key)mutable.value=state.value.copy(phase="AI answer withheld",problem=ex.message,aiStartedMs=null)
        } finally {(provider as? AutoCloseable)?.close()}
    }
}
